import { CardGrid } from "@astrojs/starlight/components";
import KitCard from "../../components/KitCard.astro";

<CardGrid>
	<KitCard title="store-kit" icon="database" href="/store-kit/overview/">
		Typed Zustand factory with persistence, versioned migrations, devtools, and reset. **1.2 KB gzipped.**
	</KitCard>
	<KitCard title="fetch-kit" icon="cloud-download" href="/fetch-kit/overview/">
		Typed HTTP client + React hooks. Retries, timeouts, response caching, dedupe, GraphQL, schema validation, abort-on-unmount. **3 KB core, 0.75 KB hooks.**
	</KitCard>
	<KitCard title="log-kit" icon="document" href="/log-kit/overview/">
		Structured logger with pluggable transports (console, HTTP, file, Datadog), perf markers, named scopes, and runtime transports. **~1.8 KB core.**
	</KitCard>
	<KitCard title="config-kit" icon="setting" href="/config-kit/overview/">
		Typed config from .env, process.env, remote sources, and module-based config files (`app.config.ts`). Schema-validated, secret-safe errors. **~2 KB.**
	</KitCard>
	<KitCard title="Tiny by design" icon="rocket">
		Bundle budgets enforced in CI. No accidental growth. No deps in the hot path - just `zustand`, `zod` (optional), and your `fetch`.
	</KitCard>
	<KitCard title="Strict TypeScript" icon="approve-check">
		`noUncheckedIndexedAccess`, `exactOptionalPropertyTypes`, the works. The API infers - you rarely write a generic.
	</KitCard>
